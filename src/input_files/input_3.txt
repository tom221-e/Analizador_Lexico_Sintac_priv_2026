BOOLEAN: es_mayor, es_igual, es_menor
INT: x, y

PROGRAM
    x = 10
    y = 5
    es_mayor = x > y
    PRINT(es_mayor)
    es_igual = x == y
    PRINT(es_igual)
    es_menor = x < y
    PRINT(es_menor)
    es_mayor = true
    PRINT(es_mayor)
    es_igual = false
    PRINT(es_igual)

% OUTPUT ESPERADO
% 1
% 0
% 0
% 1
% 0
